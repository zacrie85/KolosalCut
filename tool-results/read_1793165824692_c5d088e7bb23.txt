    73→const clampX = (v: number) => Math.min(0.98, Math.max(0.02, v));

    76→/**
    77→ * PositionPad — pengganti pilihan atas/tengah/bawah.
    78→ * Posisi teks digeser BEBAS ke atas, bawah, kiri, dan kanan lewat:
    79→ * 1. drag titik di mini-pad 9:16, atau
    80→ * 2. tombol panah untuk geser halus.
    81→ */
    82→function PositionPad({
    83→  el,
    84→  onChange,
    85→  disabled,
    86→}: {
    87→  el: TextElement;
    88→  onChange: (patch: Partial<TextElement>) => void;
    89→  disabled?: boolean;
    90→}) {
    91→  const padRef = useRef<HTMLDivElement>(null);
    92→  const draggingRef = useRef(false);
    93→
    94→  const applyFromPointer = (e: React.PointerEvent<HTMLDivElement>) => {
    95→    const pad = padRef.current;
    96→    if (!pad) return;
    97→    const rect = pad.getBoundingClientRect();
    98→    const relX = clampX((e.clientX - rect.left) / rect.width);
    99→    const relY = clampY((e.clientY - rect.top) / rect.height);
   100→    onChange({ relX, relY });
   101→  };
   102→
   103→  const onPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   104→    if (disabled) return;
   105→    e.preventDefault();
   106→    draggingRef.current = true;
   107→    e.currentTarget.setPointerCapture(e.pointerId);
   108→    applyFromPointer(e);
   109→  };
   110→
   111→  const onPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   112→    if (!draggingRef.current || disabled) return;
   113→    applyFromPointer(e);
   114→  };
   115→
   116→  const endDrag = () => {
   117→    draggingRef.current = false;
   118→  };
   119→
   120→  const nudge = (dx: number, dy: number) =>
   121→    onChange({ relX: clampX(el.relX + dx), relY: clampY(el.relY + dy) });
   122→
   123→  const arrowCls =
   124→    "h-8 w-8 p-0 border-cyan-300/25 text-zinc-300 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40";
   125→
   126→  return (
   127→    <div className="flex items-center gap-3">
   128→      {/* mini-pad 9:16 — drag bebas */}
   129→      <div
   130→        ref={padRef}
   131→        role="slider"
   132→        aria-label="Geser posisi teks (atas, bawah, kiri, kanan)"
   133→        aria-valuemin={0}
   134→        aria-valuemax={100}
   135→        aria-valuenow={Math.round(el.relX * 100)}
   136→        aria-valuetext={`horizontal ${Math.round(el.relX * 100)}%, vertikal ${Math.round(el.relY * 100)}%`}
   137→        tabIndex={disabled ? -1 : 0}
   138→        onKeyDown={(e) => {
   139→          if (disabled) return;
   140→          const step = e.shiftKey ? 0.06 : 0.02;
   141→          if (e.key === "ArrowUp") {
   142→            e.preventDefault();
   143→            nudge(0, -step);
   144→          } else if (e.key === "ArrowDown") {
   145→            e.preventDefault();
   146→            nudge(0, step);
   147→          } else if (e.key === "ArrowLeft") {
   148→            e.preventDefault();
   149→            nudge(-step, 0);
   150→          } else if (e.key === "ArrowRight") {
   151→            e.preventDefault();
   152→            nudge(step, 0);
   153→          }
   154→        }}
   155→        onPointerDown={onPointerDown}
   156→        onPointerMove={onPointerMove}
   157→        onPointerUp={endDrag}
   158→        onPointerCancel={endDrag}
   159→        className={`relative shrink-0 w-[72px] aspect-[9/16] rounded-lg border bg-slate-950 touch-none select-none ${
   160→          disabled ? "border-zinc-800 opacity-50" : "border-cyan-300/35 cursor-grab active:cursor-grabbing shadow-[0_0_14px_-4px_rgba(34,211,238,0.5)]"
   161→        }`}
   162→        style={{
   163→          backgroundImage:
   164→            "linear-gradient(#27272a 1px, transparent 1px), linear-gradient(90deg, #27272a 1px, transparent 1px)",
   165→          backgroundSize: "25% 25%",
   166→        }}
   167→      >
   168→        {/* garis tengah horizontal sebagai panduan */}
   169→        <div className="absolute inset-x-1 top-1/2 border-t border-dashed border-cyan-300/15 pointer-events-none" />
   170→        {/* titik posisi teks */}
   171→        <div
   172→          className="absolute w-5 h-5 rounded-full bg-cyan-400 border-2 border-slate-950 shadow-[0_0_12px_rgba(34,211,238,0.8)] pointer-events-none"
   173→          style={{
   174→            left: `${el.relX * 100}%`,
   175→            top: `${el.relY * 100}%`,
   176→            transform: "translate(-50%, -50%)",
   177→          }}
   178→        />
   179→      </div>
   180→
   181→      {/* tombol panah: geser halus atas bawah kiri kanan */}
   182→      <div className="flex flex-col items-center gap-1">
   183→        <Button
   184→          type="button"
   185→          size="sm"
   186→          variant="outline"
   187→          disabled={disabled}
   188→          className={arrowCls}
   189→          aria-label="Geser ke atas"
   190→          onClick={() => nudge(0, -0.02)}
   191→        >
   192→          <ChevronUp className="w-4 h-4" />
   193→        </Button>
   194→        <div className="flex gap-1">
   195→          <Button
   196→            type="button"
   197→            size="sm"
   198→            variant="outline"
   199→            disabled={disabled}
   200→            className={arrowCls}
   201→            aria-label="Geser ke kiri"
   202→            onClick={() => nudge(-0.02, 0)}
   203→          >
   204→            <ChevronLeft className="w-4 h-4" />
   205→          </Button>
   206→          <Button
   207→            type="button"
   208→            size="sm"
   209→            variant="outline"
   210→            disabled={disabled}
   211→            className="h-8 w-8 p-0 border-cyan-300/25 text-zinc-400 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40"
   212→            aria-label="Pusatkan horizontal"
   213→            title="Pusatkan horizontal"
   214→            onClick={() => onChange({ relX: 0.5 })}
   215→          >
   216→            <RotateCcw className="w-3.5 h-3.5" />
   217→          </Button>
   218→          <Button
   219→            type="button"
   220→            size="sm"
   221→            variant="outline"
   222→            disabled={disabled}
   223→            className={arrowCls}
   224→            aria-label="Geser ke kanan"
   225→            onClick={() => nudge(0.02, 0)}
   226→          >
   227→            <ChevronRight className="w-4 h-4" />
   228→          </Button>
   229→        </div>
   230→        <Button
   231→          type="button"
   232→          size="sm"
   233→          variant="outline"
   234→          disabled={disabled}
   235→          className={arrowCls}
   236→          aria-label="Geser ke bawah"
   237→          onClick={() => nudge(0, 0.02)}
   238→        >
   239→          <ChevronDown className="w-4 h-4" />
   240→        </Button>
   241→      </div>
   242→
   243→      {/* readout posisi */}
   244→      <div className="flex-1 min-w-0">
   245→        <div className="flex items-center gap-1.5 text-[11px] text-zinc-400">
   246→          <Move className="w-3.5 h-3.5 text-cyan-300 shrink-0" aria-hidden />
   247→          <span className="tabular-nums">
   248→            X {Math.round(el.relX * 100)}% · Y {Math.round(el.relY * 100)}%
   249→          </span>
   250→        </div>
   251→        <p className="text-[11px] text-zinc-500 leading-snug mt-1">
   252→          Geser titik di kotak mini (bebas semua arah) atau pakai tombol panah. Bisa juga geser
   253→          langsung teksnya di pratinjau video.
   254→        </p>
   255→      </div>
   256→    </div>
   257→  );
   258→}
