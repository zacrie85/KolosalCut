    76→    76→/**
    77→    77→ * PositionPad — pengganti pilihan atas/tengah/bawah.
    78→    78→ * Posisi teks digeser BEBAS ke atas, bawah, kiri, dan kanan lewat:
    79→    79→ * 1. drag titik di mini-pad 9:16, atau
    80→    80→ * 2. tombol panah untuk geser halus.

    82→    82→function PositionPad({
    83→    83→  el,
    84→    84→  onChange,
    85→    85→  disabled,
    86→    86→}: {
    87→    87→  el: TextElement;
    88→    88→  onChange: (patch: Partial<TextElement>) => void;
    89→    89→  disabled?: boolean;
    90→    90→}) {
    91→    91→  const padRef = useRef<HTMLDivElement>(null);
    92→    92→  const draggingRef = useRef(false);
    93→    93→
    94→    94→  const applyFromPointer = (e: React.PointerEvent<HTMLDivElement>) => {
    95→    95→    const pad = padRef.current;
    96→    96→    if (!pad) return;
    97→    97→    const rect = pad.getBoundingClientRect();
    98→    98→    const relX = clampX((e.clientX - rect.left) / rect.width);
    99→    99→    const relY = clampY((e.clientY - rect.top) / rect.height);
   100→   100→    onChange({ relX, relY });
   101→   101→  };
   102→   102→
   103→   103→  const onPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   104→   104→    if (disabled) return;
   105→   105→    e.preventDefault();
   106→   106→    draggingRef.current = true;
   107→   107→    e.currentTarget.setPointerCapture(e.pointerId);
   108→   108→    applyFromPointer(e);
   109→   109→  };
   110→   110→
   111→   111→  const onPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   112→   112→    if (!draggingRef.current || disabled) return;
   113→   113→    applyFromPointer(e);
   114→   114→  };
   115→   115→
   116→   116→  const endDrag = () => {
   117→   117→    draggingRef.current = false;
   118→   118→  };
   119→   119→
   120→   120→  const nudge = (dx: number, dy: number) =>
   121→   121→    onChange({ relX: clampX(el.relX + dx), relY: clampY(el.relY + dy) });
   122→   122→
   123→   123→  const arrowCls =
   124→   124→    "h-8 w-8 p-0 border-cyan-300/25 text-zinc-300 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40";
   125→   125→
   126→   126→  return (
   127→   127→    <div className="flex items-center gap-3">
   128→   128→      {/* mini-pad 9:16 — drag bebas */}
   129→   129→      <div
   130→   130→        ref={padRef}
   131→   131→        role="slider"
   132→   132→        aria-label="Geser posisi teks (atas, bawah, kiri, kanan)"
   133→   133→        aria-valuemin={0}
   134→   134→        aria-valuemax={100}
   135→   135→        aria-valuenow={Math.round(el.relX * 100)}
   136→   136→        aria-valuetext={`horizontal ${Math.round(el.relX * 100)}%, vertikal ${Math.round(el.relY * 100)}%`}
   137→   137→        tabIndex={disabled ? -1 : 0}
   138→   138→        onKeyDown={(e) => {
   139→   139→          if (disabled) return;
   140→   140→          const step = e.shiftKey ? 0.06 : 0.02;
   141→   141→          if (e.key === "ArrowUp") {
   142→   142→            e.preventDefault();
   143→   143→            nudge(0, -step);
   144→   144→          } else if (e.key === "ArrowDown") {
   145→   145→            e.preventDefault();
   146→   146→            nudge(0, step);
   147→   147→          } else if (e.key === "ArrowLeft") {
   148→   148→            e.preventDefault();
   149→   149→            nudge(-step, 0);
   150→   150→          } else if (e.key === "ArrowRight") {
   151→   151→            e.preventDefault();
   152→   152→            nudge(step, 0);
   153→   153→          }
   154→   154→        }}
   155→   155→        onPointerDown={onPointerDown}
   156→   156→        onPointerMove={onPointerMove}
   157→   157→        onPointerUp={endDrag}
   158→   158→        onPointerCancel={endDrag}
   159→   159→        className={`relative shrink-0 w-[72px] aspect-[9/16] rounded-lg border bg-slate-950 touch-none select-none ${
   160→   160→          disabled ? "border-zinc-800 opacity-50" : "border-cyan-300/35 cursor-grab active:cursor-grabbing shadow-[0_0_14px_-4px_rgba(34,211,238,0.5)]"
   161→   161→        }`}
   162→   162→        style={{
   163→   163→          backgroundImage:
   164→   164→            "linear-gradient(#27272a 1px, transparent 1px), linear-gradient(90deg, #27272a 1px, transparent 1px)",
   165→   165→          backgroundSize: "25% 25%",
   166→   166→        }}
   167→   167→      >
   168→   168→        {/* garis tengah horizontal sebagai panduan */}
   169→   169→        <div className="absolute inset-x-1 top-1/2 border-t border-dashed border-cyan-300/15 pointer-events-none" />
   170→   170→        {/* titik posisi teks */}
   171→   171→        <div
   172→   172→          className="absolute w-5 h-5 rounded-full bg-cyan-400 border-2 border-slate-950 shadow-[0_0_12px_rgba(34,211,238,0.8)] pointer-events-none"
   173→   173→          style={{
   174→   174→            left: `${el.relX * 100}%`,
   175→   175→            top: `${el.relY * 100}%`,
   176→   176→            transform: "translate(-50%, -50%)",
   177→   177→          }}
   178→   178→        />
   179→   179→      </div>
   180→   180→
   181→   181→      {/* tombol panah: geser halus atas bawah kiri kanan */}
   182→   182→      <div className="flex flex-col items-center gap-1">
   183→   183→        <Button
   184→   184→          type="button"
   185→   185→          size="sm"
   186→   186→          variant="outline"
   187→   187→          disabled={disabled}
   188→   188→          className={arrowCls}
   189→   189→          aria-label="Geser ke atas"
   190→   190→          onClick={() => nudge(0, -0.02)}
   191→   191→        >
   192→   192→          <ChevronUp className="w-4 h-4" />
   193→   193→        </Button>
   194→   194→        <div className="flex gap-1">
   195→   195→          <Button
   196→   196→            type="button"
   197→   197→            size="sm"
   198→   198→            variant="outline"
   199→   199→            disabled={disabled}
   200→   200→            className={arrowCls}
   201→   201→            aria-label="Geser ke kiri"
   202→   202→            onClick={() => nudge(-0.02, 0)}
   203→   203→          >
   204→   204→            <ChevronLeft className="w-4 h-4" />
   205→   205→          </Button>
   206→   206→          <Button
   207→   207→            type="button"
   208→   208→            size="sm"
   209→   209→            variant="outline"
   210→   210→            disabled={disabled}
   211→   211→            className="h-8 w-8 p-0 border-cyan-300/25 text-zinc-400 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40"
   212→   212→            aria-label="Pusatkan horizontal"
   213→   213→            title="Pusatkan horizontal"
   214→   214→            onClick={() => onChange({ relX: 0.5 })}
   215→   215→          >
   216→   216→            <RotateCcw className="w-3.5 h-3.5" />
   217→   217→          </Button>
   218→   218→          <Button
   219→   219→            type="button"
   220→   220→            size="sm"
   221→   221→            variant="outline"
   222→   222→            disabled={disabled}
   223→   223→            className={arrowCls}
   224→   224→            aria-label="Geser ke kanan"
   225→   225→            onClick={() => nudge(0.02, 0)}
   226→   226→          >
   227→   227→            <ChevronRight className="w-4 h-4" />
   228→   228→          </Button>
   229→   229→        </div>
   230→   230→        <Button
   231→   231→          type="button"
   232→   232→          size="sm"
   233→   233→          variant="outline"
   234→   234→          disabled={disabled}
   235→   235→          className={arrowCls}
   236→   236→          aria-label="Geser ke bawah"
   237→   237→          onClick={() => nudge(0, 0.02)}
   238→   238→        >
   239→   239→          <ChevronDown className="w-4 h-4" />
   240→   240→        </Button>
   241→   241→      </div>
   242→   242→
   243→   243→      {/* readout posisi */}
   244→   244→      <div className="flex-1 min-w-0">
   245→   245→        <div className="flex items-center gap-1.5 text-[11px] text-zinc-400">
   246→   246→          <Move className="w-3.5 h-3.5 text-cyan-300 shrink-0" aria-hidden />
   247→   247→          <span className="tabular-nums">
   248→   248→            X {Math.round(el.relX * 100)}% · Y {Math.round(el.relY * 100)}%
   249→   249→          </span>
   250→   250→        </div>
   251→   251→        <p className="text-[11px] text-zinc-500 leading-snug mt-1">
   252→   252→          Geser titik di kotak mini (bebas semua arah) atau pakai tombol panah. Bisa juga geser
   253→   253→          langsung teksnya di pratinjau video.
   254→   254→        </p>
   255→   255→      </div>
   256→   256→    </div>
   257→   257→  );
   258→   258→}
