    87→const clampX = (v: number) => Math.min(0.98, Math.max(0.02, v));

    90→/**
    91→ * PositionPad — pengganti pilihan atas/tengah/bawah.
    92→ * Posisi teks digeser BEBAS ke atas, bawah, kiri, dan kanan lewat:
    93→ * 1. drag titik di mini-pad 9:16, atau
    94→ * 2. tombol panah untuk geser halus.
    95→ */
    96→function PositionPad({
    97→  el,
    98→  onChange,
    99→  disabled,
   100→}: {
   101→  el: TextElement;
   102→  onChange: (patch: Partial<TextElement>) => void;
   103→  disabled?: boolean;
   104→}) {
   105→  const padRef = useRef<HTMLDivElement>(null);
   106→  const draggingRef = useRef(false);
   107→
   108→  const applyFromPointer = (e: React.PointerEvent<HTMLDivElement>) => {
   109→    const pad = padRef.current;
   110→    if (!pad) return;
   111→    const rect = pad.getBoundingClientRect();
   112→    const relX = clampX((e.clientX - rect.left) / rect.width);
   113→    const relY = clampY((e.clientY - rect.top) / rect.height);
   114→    onChange({ relX, relY });
   115→  };
   116→
   117→  const onPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   118→    if (disabled) return;
   119→    e.preventDefault();
   120→    draggingRef.current = true;
   121→    e.currentTarget.setPointerCapture(e.pointerId);
   122→    applyFromPointer(e);
   123→  };
   124→
   125→  const onPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   126→    if (!draggingRef.current || disabled) return;
   127→    applyFromPointer(e);
   128→  };
   129→
   130→  const endDrag = () => {
   131→    draggingRef.current = false;
   132→  };
   133→
   134→  const nudge = (dx: number, dy: number) =>
   135→    onChange({ relX: clampX(el.relX + dx), relY: clampY(el.relY + dy) });
   136→
   137→  const arrowCls =
   138→    "h-8 w-8 p-0 border-cyan-300/25 text-zinc-300 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40";
   139→
   140→  return (
   141→    <div className="flex items-center gap-3">
   142→      {/* mini-pad 9:16 — drag bebas */}
   143→      <div
   144→        ref={padRef}
   145→        role="slider"
   146→        aria-label="Geser posisi teks (atas, bawah, kiri, kanan)"
   147→        aria-valuemin={0}
   148→        aria-valuemax={100}
   149→        aria-valuenow={Math.round(el.relX * 100)}
   150→        aria-valuetext={`horizontal ${Math.round(el.relX * 100)}%, vertikal ${Math.round(el.relY * 100)}%`}
   151→        tabIndex={disabled ? -1 : 0}
   152→        onKeyDown={(e) => {
   153→          if (disabled) return;
   154→          const step = e.shiftKey ? 0.06 : 0.02;
   155→          if (e.key === "ArrowUp") {
   156→            e.preventDefault();
   157→            nudge(0, -step);
   158→          } else if (e.key === "ArrowDown") {
   159→            e.preventDefault();
   160→            nudge(0, step);
   161→          } else if (e.key === "ArrowLeft") {
   162→            e.preventDefault();
   163→            nudge(-step, 0);
   164→          } else if (e.key === "ArrowRight") {
   165→            e.preventDefault();
   166→            nudge(step, 0);
   167→          }
   168→        }}
   169→        onPointerDown={onPointerDown}
   170→        onPointerMove={onPointerMove}
   171→        onPointerUp={endDrag}
   172→        onPointerCancel={endDrag}
   173→        className={`relative shrink-0 w-[72px] aspect-[9/16] rounded-lg border bg-slate-950 touch-none select-none ${
   174→          disabled ? "border-zinc-800 opacity-50" : "border-cyan-300/35 cursor-grab active:cursor-grabbing shadow-[0_0_14px_-4px_rgba(34,211,238,0.5)]"
   175→        }`}
   176→        style={{
   177→          backgroundImage:
   178→            "linear-gradient(#27272a 1px, transparent 1px), linear-gradient(90deg, #27272a 1px, transparent 1px)",
   179→          backgroundSize: "25% 25%",
   180→        }}
   181→      >
   182→        {/* garis tengah horizontal sebagai panduan */}
   183→        <div className="absolute inset-x-1 top-1/2 border-t border-dashed border-cyan-300/15 pointer-events-none" />
   184→        {/* titik posisi teks */}
   185→        <div
   186→          className="absolute w-5 h-5 rounded-full bg-cyan-400 border-2 border-slate-950 shadow-[0_0_12px_rgba(34,211,238,0.8)] pointer-events-none"
   187→          style={{
   188→            left: `${el.relX * 100}%`,
   189→            top: `${el.relY * 100}%`,
   190→            transform: "translate(-50%, -50%)",
   191→          }}
   192→        />
   193→      </div>
   194→
   195→      {/* tombol panah: geser halus atas bawah kiri kanan */}
   196→      <div className="flex flex-col items-center gap-1">
   197→        <Button
   198→          type="button"
   199→          size="sm"
   200→          variant="outline"
   201→          disabled={disabled}
   202→          className={arrowCls}
   203→          aria-label="Geser ke atas"
   204→          onClick={() => nudge(0, -0.02)}
   205→        >
   206→          <ChevronUp className="w-4 h-4" />
   207→        </Button>
   208→        <div className="flex gap-1">
   209→          <Button
   210→            type="button"
   211→            size="sm"
   212→            variant="outline"
   213→            disabled={disabled}
   214→            className={arrowCls}
   215→            aria-label="Geser ke kiri"
   216→            onClick={() => nudge(-0.02, 0)}
   217→          >
   218→            <ChevronLeft className="w-4 h-4" />
   219→          </Button>
   220→          <Button
   221→            type="button"
   222→            size="sm"
   223→            variant="outline"
   224→            disabled={disabled}
   225→            className="h-8 w-8 p-0 border-cyan-300/25 text-zinc-400 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40"
   226→            aria-label="Pusatkan horizontal"
   227→            title="Pusatkan horizontal"
   228→            onClick={() => onChange({ relX: 0.5 })}
   229→          >
   230→            <RotateCcw className="w-3.5 h-3.5" />
   231→          </Button>
   232→          <Button
   233→            type="button"
   234→            size="sm"
   235→            variant="outline"
   236→            disabled={disabled}
   237→            className={arrowCls}
   238→            aria-label="Geser ke kanan"
   239→            onClick={() => nudge(0.02, 0)}
   240→          >
   241→            <ChevronRight className="w-4 h-4" />
   242→          </Button>
   243→        </div>
   244→        <Button
   245→          type="button"
   246→          size="sm"
   247→          variant="outline"
   248→          disabled={disabled}
   249→          className={arrowCls}
   250→          aria-label="Geser ke bawah"
   251→          onClick={() => nudge(0, 0.02)}
   252→        >
   253→          <ChevronDown className="w-4 h-4" />
   254→        </Button>
   255→      </div>
   256→
   257→      {/* readout posisi */}
   258→      <div className="flex-1 min-w-0">
   259→        <div className="flex items-center gap-1.5 text-[11px] text-zinc-400">
   260→          <Move className="w-3.5 h-3.5 text-cyan-300 shrink-0" aria-hidden />
   261→          <span className="tabular-nums">
   262→            X {Math.round(el.relX * 100)}% · Y {Math.round(el.relY * 100)}%
   263→          </span>
   264→        </div>
   265→        <p className="text-[11px] text-zinc-500 leading-snug mt-1">
   266→          Geser titik di kotak mini (bebas semua arah) atau pakai tombol panah. Bisa juga geser
   267→          langsung teksnya di pratinjau video.
   268→        </p>
   269→      </div>
   270→    </div>
   271→  );
   272→}
