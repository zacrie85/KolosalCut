    78→    78→/**
    79→    79→ * PositionPad — pengganti pilihan atas/tengah/bawah.
    80→    80→ * Posisi teks digeser BEBAS ke atas, bawah, kiri, dan kanan lewat:
    81→    81→ * 1. drag titik di mini-pad 9:16, atau
    82→    82→ * 2. tombol panah untuk geser halus.

    84→    84→function PositionPad({
    85→    85→  el,
    86→    86→  onChange,
    87→    87→  disabled,
    88→    88→}: {
    89→    89→  el: TextElement;
    90→    90→  onChange: (patch: Partial<TextElement>) => void;
    91→    91→  disabled?: boolean;
    92→    92→}) {
    93→    93→  const padRef = useRef<HTMLDivElement>(null);
    94→    94→  const draggingRef = useRef(false);
    95→    95→
    96→    96→  const applyFromPointer = (e: React.PointerEvent<HTMLDivElement>) => {
    97→    97→    const pad = padRef.current;
    98→    98→    if (!pad) return;
    99→    99→    const rect = pad.getBoundingClientRect();
   100→   100→    const relX = clampX((e.clientX - rect.left) / rect.width);
   101→   101→    const relY = clampY((e.clientY - rect.top) / rect.height);
   102→   102→    onChange({ relX, relY });
   103→   103→  };
   104→   104→
   105→   105→  const onPointerDown = (e: React.PointerEvent<HTMLDivElement>) => {
   106→   106→    if (disabled) return;
   107→   107→    e.preventDefault();
   108→   108→    draggingRef.current = true;
   109→   109→    e.currentTarget.setPointerCapture(e.pointerId);
   110→   110→    applyFromPointer(e);
   111→   111→  };
   112→   112→
   113→   113→  const onPointerMove = (e: React.PointerEvent<HTMLDivElement>) => {
   114→   114→    if (!draggingRef.current || disabled) return;
   115→   115→    applyFromPointer(e);
   116→   116→  };
   117→   117→
   118→   118→  const endDrag = () => {
   119→   119→    draggingRef.current = false;
   120→   120→  };
   121→   121→
   122→   122→  const nudge = (dx: number, dy: number) =>
   123→   123→    onChange({ relX: clampX(el.relX + dx), relY: clampY(el.relY + dy) });
   124→   124→
   125→   125→  const arrowCls =
   126→   126→    "h-8 w-8 p-0 border-cyan-300/25 text-zinc-300 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40";
   127→   127→
   128→   128→  return (
   129→   129→    <div className="flex items-center gap-3">
   130→   130→      {/* mini-pad 9:16 — drag bebas */}
   131→   131→      <div
   132→   132→        ref={padRef}
   133→   133→        role="slider"
   134→   134→        aria-label="Geser posisi teks (atas, bawah, kiri, kanan)"
   135→   135→        aria-valuemin={0}
   136→   136→        aria-valuemax={100}
   137→   137→        aria-valuenow={Math.round(el.relX * 100)}
   138→   138→        aria-valuetext={`horizontal ${Math.round(el.relX * 100)}%, vertikal ${Math.round(el.relY * 100)}%`}
   139→   139→        tabIndex={disabled ? -1 : 0}
   140→   140→        onKeyDown={(e) => {
   141→   141→          if (disabled) return;
   142→   142→          const step = e.shiftKey ? 0.06 : 0.02;
   143→   143→          if (e.key === "ArrowUp") {
   144→   144→            e.preventDefault();
   145→   145→            nudge(0, -step);
   146→   146→          } else if (e.key === "ArrowDown") {
   147→   147→            e.preventDefault();
   148→   148→            nudge(0, step);
   149→   149→          } else if (e.key === "ArrowLeft") {
   150→   150→            e.preventDefault();
   151→   151→            nudge(-step, 0);
   152→   152→          } else if (e.key === "ArrowRight") {
   153→   153→            e.preventDefault();
   154→   154→            nudge(step, 0);
   155→   155→          }
   156→   156→        }}
   157→   157→        onPointerDown={onPointerDown}
   158→   158→        onPointerMove={onPointerMove}
   159→   159→        onPointerUp={endDrag}
   160→   160→        onPointerCancel={endDrag}
   161→   161→        className={`relative shrink-0 w-[72px] aspect-[9/16] rounded-lg border bg-slate-950 touch-none select-none ${
   162→   162→          disabled ? "border-zinc-800 opacity-50" : "border-cyan-300/35 cursor-grab active:cursor-grabbing shadow-[0_0_14px_-4px_rgba(34,211,238,0.5)]"
   163→   163→        }`}
   164→   164→        style={{
   165→   165→          backgroundImage:
   166→   166→            "linear-gradient(#27272a 1px, transparent 1px), linear-gradient(90deg, #27272a 1px, transparent 1px)",
   167→   167→          backgroundSize: "25% 25%",
   168→   168→        }}
   169→   169→      >
   170→   170→        {/* garis tengah horizontal sebagai panduan */}
   171→   171→        <div className="absolute inset-x-1 top-1/2 border-t border-dashed border-cyan-300/15 pointer-events-none" />
   172→   172→        {/* titik posisi teks */}
   173→   173→        <div
   174→   174→          className="absolute w-5 h-5 rounded-full bg-cyan-400 border-2 border-slate-950 shadow-[0_0_12px_rgba(34,211,238,0.8)] pointer-events-none"
   175→   175→          style={{
   176→   176→            left: `${el.relX * 100}%`,
   177→   177→            top: `${el.relY * 100}%`,
   178→   178→            transform: "translate(-50%, -50%)",
   179→   179→          }}
   180→   180→        />
   181→   181→      </div>
   182→   182→
   183→   183→      {/* tombol panah: geser halus atas bawah kiri kanan */}
   184→   184→      <div className="flex flex-col items-center gap-1">
   185→   185→        <Button
   186→   186→          type="button"
   187→   187→          size="sm"
   188→   188→          variant="outline"
   189→   189→          disabled={disabled}
   190→   190→          className={arrowCls}
   191→   191→          aria-label="Geser ke atas"
   192→   192→          onClick={() => nudge(0, -0.02)}
   193→   193→        >
   194→   194→          <ChevronUp className="w-4 h-4" />
   195→   195→        </Button>
   196→   196→        <div className="flex gap-1">
   197→   197→          <Button
   198→   198→            type="button"
   199→   199→            size="sm"
   200→   200→            variant="outline"
   201→   201→            disabled={disabled}
   202→   202→            className={arrowCls}
   203→   203→            aria-label="Geser ke kiri"
   204→   204→            onClick={() => nudge(-0.02, 0)}
   205→   205→          >
   206→   206→            <ChevronLeft className="w-4 h-4" />
   207→   207→          </Button>
   208→   208→          <Button
   209→   209→            type="button"
   210→   210→            size="sm"
   211→   211→            variant="outline"
   212→   212→            disabled={disabled}
   213→   213→            className="h-8 w-8 p-0 border-cyan-300/25 text-zinc-400 hover:bg-cyan-400/10 hover:text-cyan-200 hover:border-cyan-300/60 disabled:opacity-40"
   214→   214→            aria-label="Pusatkan horizontal"
   215→   215→            title="Pusatkan horizontal"
   216→   216→            onClick={() => onChange({ relX: 0.5 })}
   217→   217→          >
   218→   218→            <RotateCcw className="w-3.5 h-3.5" />
   219→   219→          </Button>
   220→   220→          <Button
   221→   221→            type="button"
   222→   222→            size="sm"
   223→   223→            variant="outline"
   224→   224→            disabled={disabled}
   225→   225→            className={arrowCls}
   226→   226→            aria-label="Geser ke kanan"
   227→   227→            onClick={() => nudge(0.02, 0)}
   228→   228→          >
   229→   229→            <ChevronRight className="w-4 h-4" />
   230→   230→          </Button>
   231→   231→        </div>
   232→   232→        <Button
   233→   233→          type="button"
   234→   234→          size="sm"
   235→   235→          variant="outline"
   236→   236→          disabled={disabled}
   237→   237→          className={arrowCls}
   238→   238→          aria-label="Geser ke bawah"
   239→   239→          onClick={() => nudge(0, 0.02)}
   240→   240→        >
   241→   241→          <ChevronDown className="w-4 h-4" />
   242→   242→        </Button>
   243→   243→      </div>
   244→   244→
   245→   245→      {/* readout posisi */}
   246→   246→      <div className="flex-1 min-w-0">
   247→   247→        <div className="flex items-center gap-1.5 text-[11px] text-zinc-400">
   248→   248→          <Move className="w-3.5 h-3.5 text-cyan-300 shrink-0" aria-hidden />
   249→   249→          <span className="tabular-nums">
   250→   250→            X {Math.round(el.relX * 100)}% · Y {Math.round(el.relY * 100)}%
   251→   251→          </span>
   252→   252→        </div>
   253→   253→        <p className="text-[11px] text-zinc-500 leading-snug mt-1">
   254→   254→          Geser titik di kotak mini (bebas semua arah) atau pakai tombol panah. Bisa juga geser
   255→   255→          langsung teksnya di pratinjau video.
   256→   256→        </p>
   257→   257→      </div>
   258→   258→    </div>
   259→   259→  );
   260→   260→}
